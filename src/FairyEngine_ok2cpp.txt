// FairyEngine_single.cpp
#include <windows.h>
#include <string>
#include <iostream>
#include <vector>

class FairyEngine
{
public:
    FairyEngine(const std::string& stockfishPath)
        : stockfishPath(stockfishPath),
          hChildStd_IN_Wr(nullptr),
          hChildStd_OUT_Rd(nullptr),
          hProcess(nullptr)
    {}

    ~FairyEngine()
    {
        Stop();
    }

    bool Start();
    void Stop();
    bool WriteCommand(const std::string& command);
    std::string ReadLine();
    std::string GetBestMove(const std::string& fen, int depth);

private:
    std::string stockfishPath;
    HANDLE hChildStd_IN_Wr;
    HANDLE hChildStd_OUT_Rd;
    HANDLE hProcess;
    std::string outBuffer; // buffer across reads
};

// Write command (raw bytes). Returns true if all bytes written.
bool FairyEngine::WriteCommand(const std::string& command)
{
    if (!hChildStd_IN_Wr)
        return false;

    DWORD written = 0;
    BOOL ok = WriteFile(
        hChildStd_IN_Wr,
        command.c_str(),
        static_cast<DWORD>(command.size()),
        &written,
        nullptr
    );

    std::cout << "SEND CMD : " << command;
    return ok && written == command.size();
}

// ReadLine: non-blocking, uses PeekNamedPipe, accumulates in outBuffer, returns one line (without CR/LF) if available.
// If no data available returns empty string.
std::string FairyEngine::ReadLine()
{
    if (!hChildStd_OUT_Rd)
        return "";

    DWORD available = 0;
    if (!PeekNamedPipe(hChildStd_OUT_Rd, nullptr, 0, nullptr, &available, nullptr))
    {
        // error or pipe closed
        return "";
    }

    if (available == 0)
        return "";

    std::vector<char> buf(available + 1);
    DWORD read = 0;
    BOOL ok = ReadFile(hChildStd_OUT_Rd, buf.data(), available, &read, nullptr);
    if (!ok || read == 0)
        return "";

    buf[read] = '\0';
    outBuffer.append(buf.data(), read);

    // find newline (CR or LF)
    size_t pos = outBuffer.find_first_of("\r\n");
    if (pos == std::string::npos)
    {
        // no complete line yet; return entire buffer (optional behavior) or empty to wait.
        // To ensure engine responses are line-terminated, return empty here so caller loops until newline appears.
        return "";
    }

    // extract line up to pos
    std::string line = outBuffer.substr(0, pos);

    // remove the line plus any subsequent LF if CRLF
    size_t next = pos + 1;
    if (pos + 1 < outBuffer.size() && outBuffer[pos] == '\r' && outBuffer[pos + 1] == '\n')
        ++next;
    // also skip multiple consecutive CR/LF
    while (next < outBuffer.size() && (outBuffer[next] == '\r' || outBuffer[next] == '\n'))
        ++next;

    outBuffer.erase(0, next);

    std::cout << "PIPE : " << line << std::endl;
    return line;
}

bool FairyEngine::Start()
{
    SECURITY_ATTRIBUTES sa{};
    sa.nLength = sizeof(sa);
    sa.bInheritHandle = TRUE;
    sa.lpSecurityDescriptor = nullptr;

    HANDLE outWrite = nullptr;
    HANDLE inRead = nullptr;

    // create stdout pipe (child -> parent)
    if (!CreatePipe(&hChildStd_OUT_Rd, &outWrite, &sa, 0))
    {
        std::cout << "CreatePipe stdout failed, error=" << GetLastError() << "\n";
        return false;
    }
    // create stdin pipe (parent -> child)
    if (!CreatePipe(&inRead, &hChildStd_IN_Wr, &sa, 0))
    {
        std::cout << "CreatePipe stdin failed, error=" << GetLastError() << "\n";
        CloseHandle(hChildStd_OUT_Rd);
        hChildStd_OUT_Rd = nullptr;
        return false;
    }

    // ensure parent read handle is not inheritable
    if (!SetHandleInformation(hChildStd_OUT_Rd, HANDLE_FLAG_INHERIT, 0))
    {
        // not fatal, but log
        std::cout << "SetHandleInformation failed, error=" << GetLastError() << "\n";
    }

    STARTUPINFOA si{};
    PROCESS_INFORMATION pi{};
    si.cb = sizeof(si);
    si.dwFlags = STARTF_USESTDHANDLES;
    si.hStdOutput = outWrite;
    si.hStdError = outWrite;
    si.hStdInput = inRead;

    std::string cmd = "\"" + stockfishPath + "\"";

    BOOL ok = CreateProcessA(
        nullptr,
        &cmd[0],
        nullptr,
        nullptr,
        TRUE,               // inherit handles
        CREATE_NO_WINDOW,
        nullptr,
        nullptr,
        &si,
        &pi
    );

    // parent no longer needs these thread-local handles
    CloseHandle(outWrite);
    CloseHandle(inRead);

    if (!ok)
    {
        std::cout << "CREATE PROCESS FAIL, error=" << GetLastError() << "\n";
        if (hChildStd_OUT_Rd) { CloseHandle(hChildStd_OUT_Rd); hChildStd_OUT_Rd = nullptr; }
        if (hChildStd_IN_Wr) { CloseHandle(hChildStd_IN_Wr); hChildStd_IN_Wr = nullptr; }
        return false;
    }

    hProcess = pi.hProcess;
    CloseHandle(pi.hThread);

    // initialize engine
    WriteCommand("uci\n");

    // wait uciok
    for (;;)
    {
        std::string line = ReadLine();
        if (!line.empty() && line.find("uciok") != std::string::npos)
            break;
        Sleep(10);
    }

    // set variant (example)
    WriteCommand("setoption name UCI_Variant value makruk\n");
    WriteCommand("isready\n");

    for (;;)
    {
        std::string line = ReadLine();
        if (!line.empty() && line.find("readyok") != std::string::npos)
            break;
        Sleep(10);
    }

    std::cout << "FAIRY READY\n";
    return true;
}

std::string FairyEngine::GetBestMove(const std::string& fen, int depth)
{
    std::cout << "SEND FEN : " << fen << std::endl;

    // send position and go commands
    WriteCommand(std::string("position fen ") + fen + "\n");
    Sleep(20); // small pause to let engine parse position
    WriteCommand(std::string("go depth ") + std::to_string(depth) + "\n");

    // wait for bestmove line
    for (;;)
    {
        std::string line = ReadLine();
        if (!line.empty())
        {
            size_t p = line.find("bestmove");
            if (p != std::string::npos)
            {
                std::string move = line.substr(p + 9);
                // trim trailing whitespace
                while (!move.empty() && (move.back() == '\n' || move.back() == '\r' || move.back() == ' '))
                    move.pop_back();
                return move;
            }
        }
        Sleep(10);
    }
    return "";
}

void FairyEngine::Stop()
{
    if (hChildStd_IN_Wr)
    {
        WriteCommand("quit\n");
        CloseHandle(hChildStd_IN_Wr);
        hChildStd_IN_Wr = nullptr;
    }

    if (hChildStd_OUT_Rd)
    {
        CloseHandle(hChildStd_OUT_Rd);
        hChildStd_OUT_Rd = nullptr;
    }

    if (hProcess)
    {
        // request termination; engine should exit after quit, but force if still alive
        TerminateProcess(hProcess, 0);
        CloseHandle(hProcess);
        hProcess = nullptr;
    }
}
